# zsh settings which depend on the local environment

# Keep a direct route to the system ls when eza aliases are active.
lso() {
	command ls "$@"
}

# Debian and Ubuntu may install bat as batcat.
if (( ! $+commands[bat] && $+commands[batcat] )); then
	alias bat='batcat'
fi

# Homebrew completions and macOS-specific tools.
if [[ "$OSTYPE" == darwin* ]] && (( $+commands[brew] )); then
	_brew_prefix="$(brew --prefix 2>/dev/null)"
	if [[ -n "$_brew_prefix" ]]; then
		[[ -d "$_brew_prefix/share/zsh-completions" ]] && fpath=("$_brew_prefix/share/zsh-completions" $fpath)
		[[ -d "$_brew_prefix/share/zsh/site-functions" ]] && fpath=("$_brew_prefix/share/zsh/site-functions" $fpath)
	fi

	_openjdk_prefix="$(brew --prefix openjdk 2>/dev/null)"
	if [[ -d "$_openjdk_prefix" ]]; then
		export JAVA_HOME="$_openjdk_prefix"
		path=("$_openjdk_prefix/bin" $path)
	fi

	_ruby_prefix="$(brew --prefix ruby 2>/dev/null)"
	if [[ -d "$_ruby_prefix" ]]; then
		path=("$_ruby_prefix/bin" $path)
		export LDFLAGS="-L$_ruby_prefix/lib${LDFLAGS:+ $LDFLAGS}"
		export CPPFLAGS="-I$_ruby_prefix/include${CPPFLAGS:+ $CPPFLAGS}"
		export PKG_CONFIG_PATH="$_ruby_prefix/lib/pkgconfig${PKG_CONFIG_PATH:+:$PKG_CONFIG_PATH}"
	fi

	if [[ -d "/Applications/Skim.app" || -d "$HOME/Applications/Skim.app" ]]; then
		export PDFVIEWER_texdoc="open -a Skim.app"
	fi

	unset _brew_prefix _openjdk_prefix _ruby_prefix
fi

# CUDA settings.
if [[ -d "/opt/cuda" ]]; then
	export CUDA_HOME="/opt/cuda"
	path+=("$CUDA_HOME/bin")
	export LD_LIBRARY_PATH="${LD_LIBRARY_PATH:+$LD_LIBRARY_PATH:}/opt/cuda/lib64:/opt/cuda/extras/CUPTI/lib64"
fi

# rustup completions.
[[ -d "$HOME/.zfunc" ]] && fpath+=("$HOME/.zfunc")

# pyenv settings.
if (( $+commands[pyenv] )); then
	export PYENV_ROOT="${PYENV_ROOT:-$HOME/.pyenv}"
	[[ -d "$PYENV_ROOT/bin" ]] && path=("$PYENV_ROOT/bin" $path)
	eval "$(pyenv init --path 2>/dev/null)"
	eval "$(pyenv init - 2>/dev/null)"
fi

# WSL-specific settings. WINHOME becomes chezmoi machine data during migration.
if [[ -r /proc/sys/kernel/osrelease ]] && [[ "${$(</proc/sys/kernel/osrelease):l}" == *microsoft* ]]; then
	export WINHOME="/mnt/c/Users/ShutoTanabashi"
	if [[ -n "${XDG_RUNTIME_DIR:-}" && ! -e "$XDG_RUNTIME_DIR/wayland-0" ]]; then
		for _wslg_socket in /mnt/wslg/runtime-dir/wayland-0(N); do
			ln -s "$_wslg_socket" "$XDG_RUNTIME_DIR/wayland-0" 2>/dev/null || true
			break
		done
		unset _wslg_socket
	fi
fi

# Prompt.
PROMPT='%F{44}%n%f %c $GITSTATUS_PROMPT%F{44}%B %# %b%f'
RPROMPT='%F{66}%3~%f %F{244}%m%f'

# Load OpenCode API key at shell startup without storing it in this file.
if [[ -z "${OPENCODE_API_KEY:-}" ]]; then
	_opencode_api_key=""
	if (( $+commands[gopass] )); then
		_opencode_api_key="$(gopass show -o dotfiles/opencode_api_key 2>/dev/null)"
	elif (( $+commands[chezmoi] )); then
		_opencode_api_key="$(chezmoi secret keyring get --service=dotfiles --user=opencode_api_key 2>/dev/null)"
	fi
	if [[ -n "$_opencode_api_key" ]]; then
		export OPENCODE_API_KEY="$_opencode_api_key"
	fi
	unset _opencode_api_key
fi

# User-local executable paths.
typeset -U path PATH
path=("$HOME/.local/bin" $path)
[[ -d "$HOME/.opencode/bin" ]] && path=("$HOME/.opencode/bin" $path)
[[ -r "$HOME/.local/bin/env" ]] && source "$HOME/.local/bin/env"
